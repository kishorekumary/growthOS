# Planner (Daily / Weekly / Monthly) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Tasks (`user_todos`, `/todos`) with a Planner that has Today / This Week / This Month checklists, optional repeating items, habits shown inline, a leftovers review sheet, and big one-tap check-off.

**Architecture:** Two new Supabase tables (`planner_items`, `planner_completions`). All date/points/visibility rules live in pure modules under `src/lib/planner/` (unit-tested with Vitest). A client-agnostic data layer (`actions.ts`) is shared by the planner page hook, the dashboard widget, QuickLog/QuickReset/briefing, and server routes. UI is client components under `src/components/planner/`: tabs on phones, three-column board on desktop.

**Tech Stack:** Next.js 14.2 App Router, React 18, TypeScript, Supabase (`@supabase/ssr`, `@supabase/supabase-js`), Tailwind 3, date-fns 3, lucide-react, Radix dropdown-menu/dialog, Vitest (new dev dependency).

**Spec:** `docs/superpowers/specs/2026-10-13-planner-design.md`

## Global Constraints

- Weeks run **Sunday–Saturday** (`weekStartsOn: 0`) everywhere in planner code.
- "Today" is the **device-local** date string `YYYY-MM-DD` from `todayStr()` in `src/lib/habitStreak.ts`. Never derive the planner's "today" on the server with `new Date()` (Vercel runs UTC).
- Points: day 10, week 20, month 40; +5 on-time bonus for one-off items completed on or before their period end. Repeating items: base only.
- Points go through the existing `user_rewards` upsert → `reward_points_log` insert → `increment_points_balance` RPC, errors logged and non-blocking.
- New migration number is **052**. Never edit an existing migration (they're applied manually in the Supabase dashboard). Do not touch the locally modified `supabase/migrations/046_user_hidden_habits.sql`.
- `user_todos` table is **not dropped** in this work.
- Habits keep all existing behavior; planner only reads habits and completes them through `completeHabit()` (`src/lib/completeHabit.ts`). Habit undo stays on the Habits page.
- No new runtime dependencies. Only dev dependency added: `vitest`.
- Dark UI conventions: `bg-white/[0.04]` rows, `text-white`/`text-slate-*`, `rounded-xl`, `cn()` from `src/lib/utils.ts`.
- Work happens in a git worktree; merge to `develop` and push when done.

```ts
import {
  addDays, addMonths, addWeeks, endOfMonth, endOfWeek, format, parseISO, startOfMonth, startOfWeek,
} from 'date-fns'
import type { PeriodType } from './types'

// Planner weeks run Sunday–Saturday. All inputs/outputs are local
// YYYY-MM-DD strings; parseISO on a bare date yields local midnight, so the
// math is calendar-based and DST-safe.
const WEEK = { weekStartsOn: 0 } as const
const fmt = (d: Date) => format(d, 'yyyy-MM-dd')

export function periodStart(type: PeriodType, date: string): string {
  const d = parseISO(date)
  if (type === 'day') return fmt(d)
  if (type === 'week') return fmt(startOfWeek(d, WEEK))
  return fmt(startOfMonth(d))
}

export function periodEnd(type: PeriodType, start: string): string {
  const d = parseISO(start)
  if (type === 'day') return fmt(d)
  if (type === 'week') return fmt(endOfWeek(d, WEEK))
  return fmt(endOfMonth(d))
}

export function shiftPeriod(type: PeriodType, start: string, n: number): string {
  const d = parseISO(start)
  if (type === 'day') return fmt(addDays(d, n))
  if (type === 'week') return fmt(addWeeks(d, n))
  return fmt(addMonths(d, n))
}

export function currentPeriods(today: string): Record<PeriodType, string> {
  return {
    day:   periodStart('day', today),
    week:  periodStart('week', today),
    month: periodStart('month', today),
  }
}

// Earliest date any current period covers — the lower bound for loading
// completions and completed items.
export function windowStart(today: string): string {
  const { week, month } = currentPeriods(today)
  return week < month ? week : month
}

const RELATIVE: Record<PeriodType, [string, string, string]> = {
  day:   ['Yesterday', 'Today', 'Tomorrow'],
  week:  ['Last week', 'This week', 'Next week'],
  month: ['Last month', 'This month', 'Next month'],
}

export function periodLabel(type: PeriodType, start: string, today: string): string {
  const current = periodStart(type, today)
  const offset = [-1, 0, 1].find(n => shiftPeriod(type, current, n) === start)
  if (offset !== undefined) return RELATIVE[type][offset + 1]
  const d = parseISO(start)
  if (type === 'day') return format(d, 'EEE, MMM d')
  if (type === 'week') return `${format(d, 'MMM d')} – ${format(parseISO(periodEnd('week', start)), 'MMM d')}`
  return format(d, 'MMMM yyyy')
}

export interface MoveTarget { label: string; type: PeriodType; start: string }

export function moveTargets(today: string): MoveTarget[] {
  const now = currentPeriods(today)
  return [
    { label: 'Today',      type: 'day',   start: now.day },
    { label: 'Tomorrow',   type: 'day',   start: shiftPeriod('day', now.day, 1) },
    { label: 'This week',  type: 'week',  start: now.week },
    { label: 'Next week',  type: 'week',  start: shiftPeriod('week', now.week, 1) },
    { label: 'This month', type: 'month', start: now.month },
    { label: 'Next month', type: 'month', start: shiftPeriod('month', now.month, 1) },
  ]
}
```

import { periodStart } from './periods'
import type { PeriodType, PlannerCompletion, PlannerItem } from './types'

type Placement = Pick<PlannerItem, 'period_type' | 'period_start' | 'repeats' | 'ended_at'>

// Ended items (dropped one-offs, stopped repeats) disappear from every list.
export function isVisibleIn(item: Placement, type: PeriodType, start: string): boolean {
  if (item.ended_at) return false
  if (item.period_type !== type) return false
  return item.repeats ? item.period_start <= start : item.period_start === start
}

export function isDoneIn(
  item: Pick<PlannerItem, 'id' | 'repeats' | 'is_completed'>,
  start: string,
  completions: Pick<PlannerCompletion, 'item_id' | 'period_start'>[],
): boolean {
  if (!item.repeats) return item.is_completed
  return completions.some(c => c.item_id === item.id && c.period_start === start)
}

// Open one-offs whose period is already over. Repeating items never count —
// a missed repeat simply resets.
export function leftovers<T extends Placement & Pick<PlannerItem, 'is_completed'>>(items: T[], today: string): T[] {
  return items.filter(i =>
    !i.repeats && !i.is_completed && !i.ended_at && i.period_start < periodStart(i.period_type, today)
  )
}
```

import { periodEnd } from './periods'
import type { PeriodType, PlannerItem } from './types'

export const BASE_POINTS: Record<PeriodType, number> = { day: 10, week: 20, month: 40 }
export const ON_TIME_BONUS = 5

// completedOn is the device-local YYYY-MM-DD the user checked the item.
// A one-off earns the bonus if finished on or before its period's last day;
// leftovers completed from the review sheet are past it, so get base only.
export function pointsFor(
  item: Pick<PlannerItem, 'period_type' | 'period_start' | 'repeats'>,
  completedOn: string,
): number {
  const base = BASE_POINTS[item.period_type]
  if (item.repeats) return base
  return completedOn <= periodEnd(item.period_type, item.period_start) ? base + ON_TIME_BONUS : base
}
```

import { parseISO } from 'date-fns'
import { periodEnd, periodStart, shiftPeriod } from './periods'
import { isDoneIn } from './visibility'
import { PERIOD_TYPES, type PeriodType, type PlannerCompletion, type PlannerItem } from './types'

export type ReminderItem = Pick<PlannerItem, 'id' | 'title' | 'period_type' | 'period_start' | 'repeats' | 'is_completed' | 'ended_at'>

// Titles a reminder on `day` should mention, day items first. Week items are
// only nagged about on Sunday (plan) and Wednesday (midweek); month items in
// the month's last 3 days. Overdue one-offs of an active type are included.
export function reminderTitles(
  items: ReminderItem[],
  completions: Pick<PlannerCompletion, 'item_id' | 'period_start'>[],
  day: string,
): string[] {
  const dow = parseISO(day).getDay()
  const monthEnd = periodEnd('month', periodStart('month', day))
  const active: Record<PeriodType, boolean> = {
    day:   true,
    week:  dow === 0 || dow === 3,
    month: day >= shiftPeriod('day', monthEnd, -2),
  }
  const titles: string[] = []
  for (const type of PERIOD_TYPES) {
    if (!active[type]) continue
    const current = periodStart(type, day)
    for (const item of items) {
      if (item.ended_at || item.period_type !== type || item.period_start > current) continue
      const open = item.repeats ? !isDoneIn(item, current, completions) : !item.is_completed
      if (open) titles.push(item.title)
    }
  }
  return titles
}

```ts
import type { SupabaseClient } from '@supabase/supabase-js'
import { currentPeriods, periodStart, windowStart } from './periods'
import { pointsFor } from './points'
import { reminderTitles, type ReminderItem } from './reminders'
import type { PeriodType, PlannerCompletion, PlannerItem } from './types'

// Works with the browser, server and admin Supabase clients alike.
export type Db = SupabaseClient

export const ITEM_COLUMNS =
  'id, user_id, title, notes, period_type, period_start, repeats, is_completed, completed_at, points_awarded, carry_count, ended_at, sort_order, created_at'
export const COMPLETION_COLUMNS = 'id, item_id, period_start, completed_at, points_awarded'

// Everything a planner screen anchored at `from` needs: items placed from
// `from` on, anything completed since `from` (rewards), every open one-off
// (leftovers can be arbitrarily old) and every repeating item (they apply
// forward from their start).
export async function fetchPlanner(db: Db, userId: string, from: string) {
  const [items, completions] = await Promise.all([
    db.from('planner_items')
      .select(ITEM_COLUMNS)
      .eq('user_id', userId)
      .is('ended_at', null)
      .or(`repeats.eq.true,is_completed.eq.false,period_start.gte.${from},completed_at.gte.${from}`)
      .order('sort_order', { ascending: true })
      .order('created_at', { ascending: true }),
    db.from('planner_completions')
      .select(COMPLETION_COLUMNS)
      .eq('user_id', userId)
      .gte('period_start', from),
  ])
  if (items.error) throw items.error
  if (completions.error) throw completions.error
  return {
    items:       (items.data ?? []) as PlannerItem[],
    completions: (completions.data ?? []) as PlannerCompletion[],
  }
}

export interface NewItem {
  title:        string
  notes?:       string | null
  period_type:  PeriodType
  period_start: string   // any date inside the target period; aligned here
  repeats?:     boolean
}

export async function insertItem(db: Db, userId: string, fields: NewItem): Promise<PlannerItem> {
  const { data, error } = await db.from('planner_items')
    .insert({
      user_id:      userId,
      title:        fields.title.trim(),
      notes:        fields.notes?.trim() || null,
      period_type:  fields.period_type,
      period_start: periodStart(fields.period_type, fields.period_start),
      repeats:      fields.repeats ?? false,
    })
    .select(ITEM_COLUMNS)
    .single()
  if (error) throw error
  return data as PlannerItem
}

export type ItemPatch = Partial<Pick<PlannerItem,
  'title' | 'notes' | 'period_type' | 'period_start' | 'repeats' | 'carry_count' | 'ended_at' | 'sort_order'>>

export async function patchItem(db: Db, id: string, patch: ItemPatch): Promise<void> {
  const { error } = await db.from('planner_items')
    .update({ ...patch, updated_at: new Date().toISOString() })
    .eq('id', id)
  if (error) throw error
}

export async function deleteItem(db: Db, id: string): Promise<void> {
  const { error } = await db.from('planner_items').delete().eq('id', id)
  if (error) throw error
}

// Same three-step write the old Tasks feature used. A failure here must
// never undo the check-off itself, so errors are logged, not thrown.
export async function adjustPoints(db: Db, userId: string, delta: number, reason: string): Promise<void> {
  if (!delta) return
  const { error: upsertErr } = await db.from('user_rewards')
    .upsert({ user_id: userId }, { onConflict: 'user_id', ignoreDuplicates: true })
  if (upsertErr) console.error('Failed to adjust planner points:', upsertErr)
  const { error: logErr } = await db.from('reward_points_log').insert({ user_id: userId, delta, reason })
  if (logErr) console.error('Failed to adjust planner points:', logErr)
  const { error: rpcErr } = await db.rpc('increment_points_balance', { p_user_id: userId, p_delta: delta })
  if (rpcErr) console.error('Failed to adjust planner points:', rpcErr)
}

// Marks `item` done for period `start` and awards points. Both write paths
// are guarded so a duplicate tap/request awards nothing the second time.
export async function completeItem(
  db: Db, userId: string, item: PlannerItem, start: string, today: string,
): Promise<{ completion: PlannerCompletion | null; points: number }> {
  const points = pointsFor(item, today)

  if (item.repeats) {
    const { data, error } = await db.from('planner_completions')
      .insert({ item_id: item.id, user_id: userId, period_start: start, points_awarded: points })
      .select(COMPLETION_COLUMNS)
      .single()
    if (error) {
      if ((error as { code?: string }).code === '23505') return { completion: null, points: 0 } // already done
      throw error
    }
    await adjustPoints(db, userId, points, `Planner: ${item.title}`)
    return { completion: data as PlannerCompletion, points }
  }

  const now = new Date().toISOString()
  const { data, error } = await db.from('planner_items')
    .update({ is_completed: true, completed_at: now, points_awarded: points, updated_at: now })
    .eq('id', item.id)
    .eq('is_completed', false)
    .select('id')
  if (error) throw error
  if (!data?.length) return { completion: null, points: 0 } // already done
  await adjustPoints(db, userId, points, `Planner: ${item.title}`)
  return { completion: null, points }
}

// Reverses completeItem using the amount actually stored at completion time.
export async function uncompleteItem(db: Db, userId: string, item: PlannerItem, start: string): Promise<number> {
  let points = 0

  if (item.repeats) {
    const { data, error } = await db.from('planner_completions')
      .delete()
      .eq('item_id', item.id)
      .eq('period_start', start)
      .select('points_awarded')
    if (error) throw error
    points = (data ?? []).reduce((s, c) => s + (c.points_awarded as number), 0)
  } else {
    const { data, error } = await db.from('planner_items')
      .update({ is_completed: false, completed_at: null, points_awarded: 0, updated_at: new Date().toISOString() })
      .eq('id', item.id)
      .eq('is_completed', true)
      .select('id')
    if (error) throw error
    points = data?.length ? item.points_awarded : 0
  }

  await adjustPoints(db, userId, -points, `Undo: ${item.title}`)
  return points
}

// Open one-offs in (or left over from before) the current day/week/month —
// what QuickReset and the opening briefing list. Returns the query builder so
// callers can hand it to useCachedQuery.
export function openNowQuery(db: Db, userId: string, today: string) {
  const now = currentPeriods(today)
  return db.from('planner_items')
    .select(ITEM_COLUMNS)
    .eq('user_id', userId)
    .is('ended_at', null)
    .eq('repeats', false)
    .eq('is_completed', false)
    .or([
      `and(period_type.eq.day,period_start.lte.${now.day})`,
      `and(period_type.eq.week,period_start.lte.${now.week})`,
      `and(period_type.eq.month,period_start.lte.${now.month})`,
    ].join(','))
    .order('period_start', { ascending: true })
    .returns<PlannerItem[]>()
}

export async function fetchReminderTitles(db: Db, userId: string, day: string): Promise<string[]> {
  const [items, done] = await Promise.all([
    db.from('planner_items')
      .select('id, title, period_type, period_start, repeats, is_completed, ended_at')
      .eq('user_id', userId)
      .is('ended_at', null)
      .or('repeats.eq.true,is_completed.eq.false')
      .lte('period_start', day),
    db.from('planner_completions')
      .select('item_id, period_start')
      .eq('user_id', userId)
      .gte('period_start', windowStart(day)),
  ])
  return reminderTitles((items.data ?? []) as ReminderItem[], done.data ?? [], day).slice(0, 10)
}

// Tailwind's `md` breakpoint. False on the server and first client render.
export function useIsDesktop(): boolean {
  const [desktop, setDesktop] = useState(false)
  useEffect(() => {
    const mq = window.matchMedia('(min-width: 768px)')
    const sync = () => setDesktop(mq.matches)
    sync()
    mq.addEventListener('change', sync)
    return () => mq.removeEventListener('change', sync)
  }, [])
  return desktop
}

import { createSupabaseBrowserClient } from '@/lib/supabase'

import { useInFlightIds } from '@/hooks/useInFlightIds'

// All planner state for one screen. Every mutation is optimistic and rolls
// back the touched row on failure.
export function usePlanner(from: string) {
  const [items, setItems]             = useState<PlannerItem[]>([])
  const [completions, setCompletions] = useState<PlannerCompletion[]>([])
  const [loading, setLoading]         = useState(true)
  const [error, setError]             = useState<string | null>(null)
  const userIdRef = useRef<string | null>(null)
  const { inFlight: busy, begin, end } = useInFlightIds()
  const today = todayStr()

    const db = createSupabaseBrowserClient()
    const { data: { session } } = await db.auth.getSession()
    const userId = session?.user?.id
    if (!userId) { setLoading(false); return }
    userIdRef.current = userId
    try {
      const res = await fetchPlanner(db, userId, from)
      setItems(res.items)
      setCompletions(res.completions)
      setError(null)
    } catch {
      setError('Could not load your planner. Check your connection and try again.')
    }
    setLoading(false)
  }, [from])

    setItems(prev => prev.map(i => (i.id === id ? next : i)))

    const userId = userIdRef.current
    if (!userId || !fields.title.trim()) return
    const tempId = `temp-${Date.now()}`
    const temp: PlannerItem = {
      id: tempId, user_id: userId, title: fields.title.trim(), notes: fields.notes ?? null,
      period_type: fields.period_type, period_start: periodStart(fields.period_type, fields.period_start),

    setItems(prev => [...prev, temp])
    try {
      const saved = await insertItem(createSupabaseBrowserClient(), userId, fields)
      replaceItem(tempId, saved)
    } catch {
      setItems(prev => prev.filter(i => i.id !== tempId))
      setError('Could not add that item.')
    }
  }

  // Returns the points delta (positive on check, negative on uncheck, 0 if
  // nothing happened) so the row can flash "+N".
  async function toggle(item: PlannerItem, start: string): Promise<number> {
    const userId = userIdRef.current
    const key = `${item.id}:${start}`
    if (!userId || item.id.startsWith('temp-') || !begin(key)) return 0
    const db = createSupabaseBrowserClient()
    const wasDone = isDoneIn(item, start, completions)
    const tempCompletionId = `temp-${key}`
    try {
      if (!wasDone) {
        if (item.repeats) {
          setCompletions(prev => [...prev, {
            id: tempCompletionId, item_id: item.id, period_start: start,
            completed_at: new Date().toISOString(), points_awarded: 0,
          }])
        } else {
          replaceItem(item.id, { ...item, is_completed: true, completed_at: new Date().toISOString() })
        }
        const { completion, points } = await completeItem(db, userId, item, start, todayStr())
        if (item.repeats && completion) {
          setCompletions(prev => prev.map(c => (c.id === tempCompletionId ? completion : c)))
        }
        if (!item.repeats && points) {
          replaceItem(item.id, { ...item, is_completed: true, completed_at: new Date().toISOString(), points_awarded: points })
        }
        return points
      }
      if (item.repeats) {
        setCompletions(prev => prev.filter(c => !(c.item_id === item.id && c.period_start === start)))
      } else {
        replaceItem(item.id, { ...item, is_completed: false, completed_at: null, points_awarded: 0 })
      }
      return -(await uncompleteItem(db, userId, item, start))
    } catch {
      // Roll back just this row.
      replaceItem(item.id, item)
      if (item.repeats) {
        const prior = completions.filter(c => c.item_id === item.id && c.period_start === start)
        setCompletions(prev => [
          ...prev.filter(c => !(c.item_id === item.id && c.period_start === start)),
          ...prior,
        ])
      }
      setError('Could not save that change.')
      return 0
    } finally {
      end(key)
    }
  }

  async function update(item: PlannerItem, patch: ItemPatch) {
    replaceItem(item.id, { ...item, ...patch })
    try {
      await patchItem(createSupabaseBrowserClient(), item.id, patch)
    } catch {
      replaceItem(item.id, item)
      setError('Could not save that change.')
    }
  }

  const move = (item: PlannerItem, type: PeriodType, start: string) =>
    update(item, { period_type: type, period_start: periodStart(type, start) })

  const carryToNow = (item: PlannerItem) =>
    update(item, { period_start: periodStart(item.period_type, today), carry_count: item.carry_count + 1 })

  const drop = (item: PlannerItem) => update(item, { ended_at: new Date().toISOString() })

  async function remove(item: PlannerItem) {
    setItems(prev => prev.filter(i => i.id !== item.id))
    try {
      await deleteItem(createSupabaseBrowserClient(), item.id)
    } catch {
      setItems(prev => [...prev, item])
      setError('Could not delete that item.')
    }
  }

  return {
    items, completions, loading, error, clearError: () => setError(null), today, busy,
    add, toggle, update, move, carryToNow, drop, remove,
  }
}

'use client'

import { useMemo } from 'react'
import { useCachedQuery } from '@/hooks/useCachedQuery'
import { useInFlightIds } from '@/hooks/useInFlightIds'
import { useReward } from '@/contexts/RewardContext'
import { completeHabit } from '@/lib/completeHabit'
import { localDateStr, todayStr } from '@/lib/habitStreak'
import { periodStart } from '@/lib/planner/periods'

interface HabitRow {
  id:           string
  habit_name:   string
  streak_count: number
  last_done_at: string | null
  frequency:    'daily' | 'weekly'
  is_global:    boolean
}
interface HabitLog { habit_id: string; status: string; log_date: string }

export interface PlannerHabit {
  id:           string
  habit_name:   string
  streak_count: number
  frequency:    'daily' | 'weekly'
  done:         boolean
}

// Today's daily habits and this week's weekly habits, in planner shape.
// Visibility mirrors TodayHabits (own + global minus hidden). Completing goes
// through completeHabit so streaks, chest and points behave exactly as on the
// Habits page; undo stays there.
export function usePlannerHabits() {
  const today     = todayStr()
  const weekStart = periodStart('week', today)

  const { data: habits, setData: setHabits } = useCachedQuery<HabitRow[]>(
    'planner:habits',
    (db, userId) => db
      .from('personality_habits')
      .select('id, habit_name, streak_count, last_done_at, frequency, is_global')
      .or(`user_id.eq.${userId},is_global.eq.true`)
      .order('created_at', { ascending: true }),
    [],
  )
  const { data: hidden } = useCachedQuery<{ habit_id: string }[]>(
    'hidden-global-habits',
    (db, userId) => db.from('user_hidden_habits').select('habit_id').eq('user_id', userId),
    [],
  )
  const { data: logs, setData: setLogs } = useCachedQuery<HabitLog[]>(
    `planner:habit-logs:${weekStart}:${today}`,
    (db, userId) => db
      .from('habit_logs')
      .select('habit_id, status, log_date')
      .eq('user_id', userId)
      .gte('log_date', weekStart)
      .lte('log_date', today),
    [],
    [weekStart, today],
  )
  const { inFlight: busy, begin, end } = useInFlightIds()
  const { celebrateMilestones } = useReward()

  const { daily, weekly } = useMemo(() => {
    const hiddenIds = new Set(hidden.map(h => h.habit_id))
    const visible = habits.filter(h => !h.is_global || !hiddenIds.has(h.id))
    const doneOn = (id: string, pred: (l: HabitLog) => boolean) =>
      logs.some(l => l.habit_id === id && l.status === 'done' && pred(l))
    const toPlanner = (h: HabitRow, done: boolean): PlannerHabit =>
      ({ id: h.id, habit_name: h.habit_name, streak_count: h.streak_count, frequency: h.frequency, done })
    return {
      daily: visible.filter(h => h.frequency !== 'weekly').map(h => toPlanner(h,
        doneOn(h.id, l => l.log_date === today) ||
        // Same fallback as TodayHabits — never for globals (shared column).
        (!h.is_global && !!h.last_done_at && localDateStr(new Date(h.last_done_at)) === today))),
      weekly: visible.filter(h => h.frequency === 'weekly').map(h => toPlanner(h, doneOn(h.id, () => true))),
    }
  }, [habits, hidden, logs, today])

  async function markDone(h: PlannerHabit) {
    if (h.done || !begin(h.id)) return
    setLogs(prev => [...prev, { habit_id: h.id, status: 'done', log_date: today }])
    try {
      const { streak_count, milestones } = await completeHabit(h.id)
      setHabits(prev => prev.map(x => (x.id === h.id
        ? { ...x, streak_count, last_done_at: new Date().toISOString() }
        : x)))
      celebrateMilestones(milestones)
    } catch {
      setLogs(prev => prev.filter(l => !(l.habit_id === h.id && l.log_date === today && l.status === 'done')))
    }
    end(h.id)
  }

  return { daily, weekly, busy, markDone }
}

  onMove:      (item: PlannerItem, type: PeriodType, start: string) => void
  onDelete:    (item: PlannerItem) => void
  onHabitDone: (habit: PlannerHabit) => void
  onDropItem?: (itemId: string) => void
}

const ADD_PLACEHOLDER: Record<PeriodType, string> = {
  day: 'Add to this day…', week: 'Add to this week…', month: 'Add to this month…',
}

export function countDone({ items, completions, habits, start }: Pick<PeriodListProps, 'items' | 'completions' | 'habits' | 'start'>) {
  const done = items.filter(i => isDoneIn(i, start, completions)).length + habits.filter(h => h.done).length
  return { done, total: items.length + habits.length }
}

export default function PeriodList(props: PeriodListProps) {
  const {
    type, start, today, variant, items, completions, habits, habitBusy, itemBusy, highlightId,
    onNavigate, onAdd, onToggle, onUpdate, onMove, onDelete, onHabitDone, onDropItem,
  } = props
  const [showDone, setShowDone] = useState(false)
  const [dragOver, setDragOver] = useState(false)

  const { done, total } = countDone(props)
  const pct = total ? Math.round((done / total) * 100) : 0
  const openItems  = items.filter(i => !isDoneIn(i, start, completions))
  const doneItems  = items.filter(i => isDoneIn(i, start, completions))
  const openHabits = habits.filter(h => !h.done)
  const doneHabits = habits.filter(h => h.done)
  const doneCount  = doneItems.length + doneHabits.length

  const renderItem = (item: PlannerItem) => (
    <PlannerItemRow
      key={item.id}
      item={item}
      done={isDoneIn(item, start, completions)}
      busy={itemBusy.has(`${item.id}:${start}`)}
      today={today}
      draggable={variant === 'column'}
      highlight={highlightId === item.id}
      onToggle={() => onToggle(item)}
      onUpdate={patch => onUpdate(item, patch)}
      onMove={(t, s) => onMove(item, t, s)}
      onDelete={() => onDelete(item)}
    />
  )

  return (
    <section
      onDragOver={onDropItem ? e => { e.preventDefault(); setDragOver(true) } : undefined}
      onDragLeave={onDropItem ? () => setDragOver(false) : undefined}
      onDrop={onDropItem ? e => {
        e.preventDefault()
        setDragOver(false)
        const id = e.dataTransfer.getData('text/plain')
        if (id) onDropItem(id)
      } : undefined}
      className={cn(
        'space-y-3 rounded-2xl transition-colors',
        variant === 'column' && 'border border-white/8 bg-white/[0.02] p-4',
        dragOver && 'border-violet-500/50 bg-violet-500/[0.05]',
      )}
    >
      {/* Header */}
      <div className="flex items-center gap-3">
        {variant === 'tab' && <ProgressRing done={done} total={total} />}
        <div className="min-w-0 flex-1">
          <div className="flex items-center gap-1">
            {onNavigate && (
              <button type="button" onClick={() => onNavigate(-1)} className="rounded-md p-1 text-slate-500 hover:text-white" aria-label="Previous">
                <ChevronLeft className="h-4 w-4" />
              </button>
            )}
            <h2 className="truncate text-base font-bold text-white">{periodLabel(type, start, today)}</h2>
            {onNavigate && (
              <button type="button" onClick={() => onNavigate(1)} className="rounded-md p-1 text-slate-500 hover:text-white" aria-label="Next">
                <ChevronRight className="h-4 w-4" />
              </button>
            )}
            <span className="ml-auto text-xs tabular-nums text-slate-500">{done} / {total}</span>
          </div>
          {variant === 'column' && (
            <div className="mt-2 h-1.5 overflow-hidden rounded-full bg-white/8">
              <div className="h-full rounded-full bg-gradient-to-r from-emerald-500 to-emerald-400 transition-all duration-500" style={{ width: `${pct}%` }} />
            </div>
          )}
        </div>
      </div>

      <QuickAdd placeholder={ADD_PLACEHOLDER[type]} onAdd={onAdd} />

      {/* Open */}
      <div className="space-y-2">
        {openHabits.map(h => (
          <HabitRow key={h.id} habit={h} busy={habitBusy.has(h.id)} onDone={() => onHabitDone(h)} />
        ))}
        {openItems.map(renderItem)}
        {openHabits.length + openItems.length === 0 && (
          <p className="py-3 text-center text-xs text-slate-600">
            {total > 0 ? 'All clear 🎉' : 'Nothing planned yet.'}
          </p>
        )}
      </div>

      {/* Done */}
      {doneCount > 0 && (
        <div className="space-y-2">
          <button
            type="button"
            onClick={() => setShowDone(s => !s)}
            className="flex items-center gap-1 text-xs font-medium text-slate-500 hover:text-slate-300"
          >
            {showDone ? <ChevronUp className="h-3.5 w-3.5" /> : <ChevronDown className="h-3.5 w-3.5" />}
            Done ({doneCount})
          </button>
          {showDone && (
            <>
              {doneHabits.map(h => (
                <HabitRow key={h.id} habit={h} busy={habitBusy.has(h.id)} onDone={() => onHabitDone(h)} />
              ))}
              {doneItems.map(renderItem)}
            </>
          )}
        </div>
      )}
    </section>
  )
}
```

Note: a just-checked row moves to the collapsed Done group on the next render; the `animate-pop` on the circle plays first because the row re-renders as done before moving. ```tsx
'use client'

import { useEffect, useMemo, useRef, useState } from 'react'
import { useSearchParams } from 'next/navigation'
import { X } from 'lucide-react'
import { cn } from '@/lib/utils'
import { todayStr } from '@/lib/habitStreak'
import { usePlanner } from '@/hooks/usePlanner'
import { usePlannerHabits } from '@/hooks/usePlannerHabits'
import { useIsDesktop } from '@/hooks/useIsDesktop'
import { currentPeriods, shiftPeriod, windowStart } from '@/lib/planner/periods'
import { isVisibleIn, leftovers } from '@/lib/planner/visibility'
import { PERIOD_TYPES, type PeriodType } from '@/lib/planner/types'
import PeriodList, { countDone } from './PeriodList'
import ReviewSheet from './ReviewSheet'
import PlannerRewards, { type DoneEntry } from './PlannerRewards'

const TAB_LABEL: Record<PeriodType, string> = { day: 'Today', week: 'Week', month: 'Month' }
const REVIEW_DISMISSED_KEY = 'planner:review-dismissed'

function reviewDismissed(): boolean {
  try { return sessionStorage.getItem(REVIEW_DISMISSED_KEY) === '1' } catch { return false }
}
function dismissReview() {
  try { sessionStorage.setItem(REVIEW_DISMISSED_KEY, '1') } catch { /* storage unavailable */ }
}

export default function PlannerView() {
  const params    = useSearchParams()
  const isDesktop = useIsDesktop()
  const habits    = usePlannerHabits()

  const [starts, setStarts] = useState(() => currentPeriods(todayStr()))
  const [tab, setTab]       = useState<PeriodType>('day')
  const [reviewOpen, setReviewOpen] = useState(false)
  const [highlightId, setHighlightId] = useState<string | null>(null)

  // Load from the earliest period on screen (or the current window).
  const from = [starts.day, starts.week, starts.month, windowStart(todayStr())].sort()[0]
  const planner = usePlanner(from)
  const { today } = planner
  const now = currentPeriods(today)

  const leftoverItems = useMemo(() => leftovers(planner.items, today), [planner.items, today])

  // Open the review once per session when leftovers exist (always when ?review=1).
  const reviewChecked = useRef(false)
  useEffect(() => {
    if (planner.loading || reviewChecked.current) return
    reviewChecked.current = true
    if (leftoverItems.length && (params.get('review') === '1' || !reviewDismissed())) setReviewOpen(true)
  }, [planner.loading, leftoverItems.length, params])

  useEffect(() => {
    if (reviewOpen && !leftoverItems.length) setReviewOpen(false)
  }, [reviewOpen, leftoverItems.length])

  // ?highlight=<id> from search: jump to the item's period and flash it.
  const highlightChecked = useRef(false)
  useEffect(() => {
    if (planner.loading || highlightChecked.current) return
    highlightChecked.current = true
    const id = params.get('highlight')
    const item = id ? planner.items.find(i => i.id === id) : undefined
    if (!item) return
    if (!item.repeats) setStarts(s => ({ ...s, [item.period_type]: item.period_start }))
    setTab(item.period_type)
    setHighlightId(item.id)
    setTimeout(() => {
      document.getElementById(`planner-item-${item.id}`)?.scrollIntoView({ behavior: 'smooth', block: 'center' })
      setTimeout(() => setHighlightId(null), 2500)
    }, 150)
  }, [planner.loading, planner.items, params])

  const rewardEntries: DoneEntry[] = useMemo(() => [
    ...planner.items
      .filter(i => i.is_completed && i.completed_at)
      .map(i => ({ completed_at: i.completed_at!, points: i.points_awarded })),
    ...planner.completions.map(c => ({ completed_at: c.completed_at, points: c.points_awarded })),
  ], [planner.items, planner.completions])

  function listProps(type: PeriodType, variant: 'column' | 'tab') {
    const start = starts[type]
    const isCurrent = start === now[type]
    const listHabits = !isCurrent ? [] : type === 'day' ? habits.daily : type === 'week' ? habits.weekly : []
    return {
      type, start, today, variant,
      items:       planner.items.filter(i => isVisibleIn(i, type, start)),
      completions: planner.completions,
      habits:      listHabits,
      habitBusy:   habits.busy,
      itemBusy:    planner.busy,
      highlightId,
      onNavigate:  (delta: number) => setStarts(s => ({ ...s, [type]: shiftPeriod(type, s[type], delta) })),
      onAdd:       (title: string, repeats: boolean) => planner.add({ title, period_type: type, period_start: start, repeats }),
      onToggle:    (item: Parameters<typeof planner.toggle>[0]) => planner.toggle(item, start),
      onUpdate:    planner.update,
      onMove:      planner.move,
      onDelete:    planner.remove,
      onHabitDone: habits.markDone,
      onDropItem:  variant === 'column'
        ? (id: string) => {
            const item = planner.items.find(i => i.id === id)
            if (item && !(item.period_type === type && item.period_start === start)) planner.move(item, type, start)
          }
        : undefined,
    }
  }

  return (
    <div className="space-y-4">
      {planner.error && (
        <div className="flex items-center gap-2 rounded-xl border border-red-500/30 bg-red-500/10 px-3 py-2 text-sm text-red-200">
          <span className="flex-1">{planner.error}</span>
          <button type="button" onClick={planner.clearError} aria-label="Dismiss"><X className="h-4 w-4" /></button>
        </div>
      )}

      {leftoverItems.length > 0 && !reviewOpen && (
        <button
          type="button"
          onClick={() => setReviewOpen(true)}
          className="w-full rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-2.5 text-left text-sm font-medium text-amber-200 hover:bg-amber-500/15"
        >
          {leftoverItems.length} leftover{leftoverItems.length === 1 ? '' : 's'} to review →
        </button>
      )}

      <PlannerRewards done={rewardEntries} />

      {isDesktop ? (
        <div className="grid grid-cols-3 gap-4">
          {PERIOD_TYPES.map(type => <PeriodList key={type} {...listProps(type, 'column')} />)}
        </div>
      ) : (
        <div className="space-y-4">
          <div className="flex gap-1 rounded-xl bg-white/5 p-1" role="tablist">
            {PERIOD_TYPES.map(type => {
              const { done, total } = countDone(listProps(type, 'tab'))
              return (
                <button
                  key={type}
                  type="button"
                  role="tab"
                  aria-selected={tab === type}
                  onClick={() => setTab(type)}
                  className={cn(
                    'flex-1 rounded-lg py-2 text-center text-sm font-semibold transition-all',
                    tab === type ? 'bg-violet-600 text-white shadow-sm' : 'text-slate-400 hover:text-white',
                  )}
                >
                  {TAB_LABEL[type]} <span className="text-xs font-normal opacity-80">{done}/{total}</span>
                </button>
              )
            })}
          </div>
          <PeriodList key={tab} {...listProps(tab, 'tab')} />
        </div>
      )}

      <ReviewSheet
        open={reviewOpen}
        onOpenChange={open => { setReviewOpen(open); if (!open) dismissReview() }}
        items={leftoverItems}
        today={today}
        onMoveToNow={planner.carryToNow}
        onReschedule={planner.move}
        onDone={item => planner.toggle(item, item.period_start)}
        onDrop={planner.drop}
        onLater={() => { dismissReview(); setReviewOpen(false) }}
      />
    </div>
  )
}

```tsx
'use client'

import Link from 'next/link'
import { ArrowRight, CheckSquare } from 'lucide-react'
import { usePlanner } from '@/hooks/usePlanner'
import { todayStr } from '@/lib/habitStreak'
import { currentPeriods, windowStart } from '@/lib/planner/periods'
import { isDoneIn, isVisibleIn, leftovers } from '@/lib/planner/visibility'
import PeriodList from './PeriodList'

const NO_HABITS: never[] = []
const NO_BUSY: ReadonlySet<string> = new Set()

// Dashboard card: today's planner list with the same one-tap rows. Habits are
// left out here because the TodayHabits card sits right above it.
export default function PlannerWidget() {
  const planner = usePlanner(windowStart(todayStr()))
  const { today } = planner
  const now = currentPeriods(today)

  const dayItems    = planner.items.filter(i => isVisibleIn(i, 'day', now.day))
  const weekOpen    = planner.items.filter(i => isVisibleIn(i, 'week', now.week) && !isDoneIn(i, now.week, planner.completions)).length
  const leftoverCnt = leftovers(planner.items, today).length

  return (
    <div className="space-y-3 rounded-2xl border border-white/8 bg-white/3 p-5">
      <div className="flex items-center gap-2">
        <CheckSquare className="h-4 w-4 text-sky-400" />
        <span className="text-sm font-semibold text-white">Today&apos;s Plan</span>
        <Link href="/planner" className="ml-auto flex items-center gap-1 text-xs text-slate-500 hover:text-violet-400">
          Open planner <ArrowRight className="h-3 w-3" />
        </Link>
      </div>

      {leftoverCnt > 0 && (
        <Link
          href="/planner?review=1"
          className="block rounded-xl border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-sm font-medium text-amber-200 hover:bg-amber-500/15"
        >
          {leftoverCnt} leftover{leftoverCnt === 1 ? '' : 's'} to review →
        </Link>
      )}

      <PeriodList
        type="day"
        start={now.day}
        today={today}
        variant="tab"
        items={dayItems}
        completions={planner.completions}
        habits={NO_HABITS}
        habitBusy={NO_BUSY}
        itemBusy={planner.busy}
        onAdd={(title, repeats) => planner.add({ title, period_type: 'day', period_start: now.day, repeats })}
        onToggle={item => planner.toggle(item, now.day)}
        onUpdate={planner.update}
        onMove={planner.move}
        onDelete={planner.remove}
        onHabitDone={() => {}}
      />

      {weekOpen > 0 && (
        <Link href="/planner" className="block text-center text-xs text-slate-500 hover:text-violet-400">
          {weekOpen} open this week →
        </Link>
      )}
    </div>
  )
}
```
